# JetaFlow — Etiquetas e Expedição Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** From a completed Ordem de Serviço, generate one shipping label per physical volume (client, OS number, per-volume internal code, QR code), and let an operator scan/type each volume's code on a conferência screen to mark it separated before dispatch.

**Architecture:** Extends the existing JetaFlow Next.js App Router app (sub-projects 1-3, already merged to `master`). Same layering as prior sub-projects: a pure calculation module (`lib/services/expedicaoCalculo.ts`) for the internal-code format and conference-progress counting; business logic + DB access in `lib/services/expedicaoService.ts`; thin API route wrappers; React pages; PDF generation via `@react-pdf/renderer` (same pattern as the existing Orçamento PDF). `Expedicao` is 1:1 with `OrdemServico` and disposable — regenerating deletes the old `Expedicao` (cascading to its `Volume` rows) and creates a fresh one.

**Tech Stack:** Same as sub-projects 1-3 (Next.js 14 App Router, TypeScript, Prisma/SQLite, Zod, Vitest, `@react-pdf/renderer`). One new dependency: `qrcode` (+ `@types/qrcode`).

**Spec:** `docs/superpowers/specs/2026-09-15-jetaflow-etiquetas-expedicao-design.md`

## Global Constraints

- `Expedicao`/`Volume` are new Prisma models — no enum fields involved here.
- Business logic lives in `lib/services/expedicaoService.ts`, unit-tested directly against the real Prisma test DB. `expedicaoCalculo.ts` is pure (no Prisma import), same pattern as `orcamentoCalculo.ts`/`ordemServicoCalculo.ts`.
- API route handlers are thin wrappers (session lookup → service call → `handleApiError`), verified manually via `npm run dev`.
- **Permission model:** any authenticated user (ADMIN or OPERADOR) can generate/regenerate etiquetas and conferir volumes — no `assertAdmin` gating anywhere in this module. Every route still requires a session (401 if `getSessionRole()` returns null).
- `codigoInterno` format is `{OS.numero}-{volumeNumero padded to 2 digits}` (e.g. `OS0001-01`), generated by `gerarCodigoInterno` — defined once in `expedicaoCalculo.ts`, never re-implemented elsewhere.
- Conference progress/state (`Volume.conferido`) never touches `OrdemServico.estagio` — the two are fully decoupled, by explicit design decision.
- `Expedicao.ordemServicoId` and `Volume.expedicaoId` use `onDelete: Cascade` — this is a deliberate contrast with `OrdemServico.orcamentoId`'s `onDelete: Restrict`: an `Expedicao` is disposable/regenerable, not a protected parent link.
- `conferirVolume` must scope its `codigoInterno` lookup to the given `expedicaoId` — a code from a different OS's expedição must never match.
- Reuses `handleApiError`, `getSessionRole`, `ForbiddenError`/`NotFoundError` (sub-project 1, unchanged) and `buscarOrdemServico` (sub-project 3, unchanged).

// lib/services/expedicaoCalculo.ts
export function gerarCodigoInterno(numeroOS: string, volumeNumero: number): string {
  return `${numeroOS}-${String(volumeNumero).padStart(2, "0")}`;
}

export function calcularProgressoConferencia(
  volumes: { conferido: boolean }[]
): { conferidos: number; total: number } {
  return {
    conferidos: volumes.filter((v) => v.conferido).length,
    total: volumes.length,
  };
}
```

// lib/services/expedicaoService.ts
import { prisma } from "@/lib/prisma";
import { buscarOrdemServico } from "@/lib/services/ordemServicoService";
import { gerarCodigoInterno } from "@/lib/services/expedicaoCalculo";
import { NotFoundError } from "@/lib/errors";
import type { ExpedicaoInput } from "@/lib/validators/expedicao";
import type { Prisma } from "@prisma/client";

const INCLUDE_VOLUMES = {
  volumes: { orderBy: { numero: "asc" } },
} satisfies Prisma.ExpedicaoInclude;

export type ExpedicaoComVolumes = Prisma.ExpedicaoGetPayload<{
  include: typeof INCLUDE_VOLUMES;
}>;

export async function gerarExpedicao(
  ordemServicoId: string,
  input: ExpedicaoInput
): Promise<ExpedicaoComVolumes> {
  const os = await buscarOrdemServico(ordemServicoId);

  const enderecoData = {
    cep: input.cep ?? null,
    endereco: input.endereco ?? null,
    numero: input.numero ?? null,
    complemento: input.complemento ?? null,
    bairro: input.bairro ?? null,
    cidade: input.cidade ?? null,
    uf: input.uf ?? null,
  };

  return prisma.$transaction(async (tx) => {
    await tx.expedicao.deleteMany({ where: { ordemServicoId } });

    const volumesData = Array.from({ length: input.totalVolumes }, (_, i) => {
      const numero = i + 1;
      return { numero, codigoInterno: gerarCodigoInterno(os.numero, numero) };
    });

    return tx.expedicao.create({
      data: {
        ordemServicoId,
        totalVolumes: input.totalVolumes,
        ...enderecoData,
        volumes: { create: volumesData },
      },
      include: INCLUDE_VOLUMES,
    });
  });
}

export async function buscarExpedicaoPorOS(ordemServicoId: string): Promise<ExpedicaoComVolumes> {
  const expedicao = await prisma.expedicao.findUnique({
    where: { ordemServicoId },
    include: INCLUDE_VOLUMES,
  });
  if (!expedicao) throw new NotFoundError();
  return expedicao;
}

export async function buscarExpedicao(id: string): Promise<ExpedicaoComVolumes> {
  const expedicao = await prisma.expedicao.findUnique({
    where: { id },
    include: INCLUDE_VOLUMES,
  });
  if (!expedicao) throw new NotFoundError();
  return expedicao;
}
```

import { NextRequest, NextResponse } from "next/server";

import { NextRequest, NextResponse } from "next/server";
import QRCode from "qrcode";

import { getSessionRole } from "@/lib/permissions";
import { handleApiError } from "@/lib/api-helpers";

      },

import { buscarExpedicao } from "@/lib/services/expedicaoService";

```tsx
"use client";

import { useRef, useState } from "react";
import { useRouter } from "next/navigation";
import { calcularProgressoConferencia } from "@/lib/services/expedicaoCalculo";

interface VolumeItem {
  id: string;
  numero: number;
  codigoInterno: string;
  conferido: boolean;
  conferidoEm: string | null;
}

export default function ConferenciaClient({
  expedicaoId,
  volumes,
}: {
  expedicaoId: string;
  volumes: VolumeItem[];
}) {
  const router = useRouter();
  const inputRef = useRef<HTMLInputElement>(null);
  const [codigo, setCodigo] = useState("");
  const [erro, setErro] = useState("");

  const progresso = calcularProgressoConferencia(volumes);

  async function bipar(e: React.FormEvent) {
    e.preventDefault();
    setErro("");
    const response = await fetch(`/api/expedicao/${expedicaoId}/conferir`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ codigoInterno: codigo }),
    });
    if (!response.ok) {
      const data = await response.json().catch(() => ({}));
      setErro(data.details?.[0]?.message ?? data.error ?? "Erro ao conferir volume");
      return;
    }
    setCodigo("");
    router.refresh();
    inputRef.current?.focus();
  }

  return (
    <div>
      <h1 className="mb-4 text-2xl font-semibold text-marinho">Conferência de Expedição</h1>

      <p className="mb-4 text-sm text-gray-600">
        {progresso.conferidos} de {progresso.total} volumes conferidos
      </p>

      <form onSubmit={bipar} className="mb-6 flex gap-2">
        <input
          ref={inputRef}
          type="text"
          autoFocus
          placeholder="Bipar ou digitar código do volume"
          value={codigo}
          onChange={(e) => setCodigo(e.target.value)}
          className="flex-1 rounded border px-3 py-2"
        />
        <button type="submit" className="rounded bg-ciano px-4 py-2 text-sm text-white">
          Conferir
        </button>
      </form>

      {erro && <p className="mb-4 text-sm text-rosa">{erro}</p>}

      <table className="w-full text-left text-sm">
        <thead>
          <tr className="border-b text-gray-500">
            <th className="py-2">Código</th>
            <th>Status</th>
            <th>Conferido em</th>
          </tr>
        </thead>
        <tbody>
          {volumes.map((volume) => (
            <tr key={volume.id} className="border-b">
              <td className="py-2">{volume.codigoInterno}</td>
              <td>
                <span
                  className={`rounded px-2 py-1 text-xs font-medium text-white ${
                    volume.conferido ? "bg-ciano" : "bg-gray-400"
                  }`}
                >
                  {volume.conferido ? "Conferido" : "Pendente"}
                </span>
              </td>
              <td>
                {volume.conferidoEm ? new Date(volume.conferidoEm).toLocaleString("pt-BR") : "—"}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```
